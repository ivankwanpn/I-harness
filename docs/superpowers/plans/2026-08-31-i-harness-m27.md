```ts
// apps/cli/src/index.ts — 新增導出 helper + web 分支改用
export function pickWebPort(args: string[], envPort: string | undefined): number {
  const idx = args.indexOf("--port")
  const flagPort = idx !== -1 ? Number(args[idx + 1]) : undefined
  return flagPort !== undefined && Number.isInteger(flagPort) && flagPort > 0
    ? flagPort
    : envPort !== undefined ? parsePort(envPort) : 4310
}
// web 分支: port: pickWebPort(args, process.env.PORT)
```

```ts
// packages/core-tools/src/context-remaining.ts
import type { ToolRegistry } from "./index.ts"
import { activeTokens } from "@i-harness/token-meter"

export function registerContextRemaining(ctx: PluginContext, tools: ToolRegistry, opts: { contextWindow?: number }): void {
  const window = opts.contextWindow
  if (window === undefined) return // fail-closed: 無窗口知識 → 不註冊
  tools.register({
    name: "get_context_remaining",
    description: "回報剩餘上下文預算（window/used/remaining/percentage）。",
    inputSchema: { type: "object" as const, properties: {}, additionalProperties: false },
    execute: async () => {
      const used = activeTokens([]) // actual: 經由 ctx.services 的 Meter 或 session 事件估算——以 core-agent 現有接入為準
      return { window, used, remaining: Math.max(0, window - used), percentage: Math.round((used / window) * 100) }
    },
  })
}
// assembly.ts: registerContextRemaining(ctx, tools, { contextWindow: assemblyModelContextWindow })
```

```ts
export function repairTurnTail(events: SessionEvent[]): SessionEvent[] {
  const out = events.map((e) => ({ ...e })) // 不把原始引用寫入 out? 合成是 append（新對象）
  let lastTurnOpen = false, lastStepOpen = false
  for (const e of events) {
    if (e.type === "turn/start") lastTurnOpen = true
    if (e.type === "turn/end") lastTurnOpen = false
    if (e.type === "step/start") lastStepOpen = true
    if (e.type === "step/end") lastStepOpen = false
  }
  // 缺失 tool/result: 最後一個未配對 tool/call → 合成 error result（M10a 語義, TOOL_ABORTED_BEFORE_DISPATCH）
  const pendingCalls = ... // 掃描秤對
  if (pendingCalls.length) out.push({ type: "tool/result", callId, name, output: { error: "TOOL_ABORTED_BEFORE_DISPATCH" } })
  if (lastStepOpen) out.push({ type: "step/end" })
  if (lastTurnOpen) out.push({ type: "turn/end" })
  return out
}
```
